# Groups of objects: objects with the same Group name stay separate shapes but move together
# (dragging one of them), the list shows the group's row, and an object outside it stays put.
action NEW_DOCUMENT
wait 800
button Box
wait 1200
button Cylinder
wait 1200
button Sphere
wait 1200
tree Box
field X=-60
field Group=Set
wait 500
tree Cylinder
field X=40
field Group=Set
wait 800
tree Sphere
field X=120
wait 800
tree Set
wait 500
expect-left Box -70
expect-left Sphere 110
button Top
wait 600
button Fit
wait 1000
drag-object Cylinder 100 0
wait 1200
expect-left Sphere 110
expect-left Box -30
expect-left Cylinder 70
quit
